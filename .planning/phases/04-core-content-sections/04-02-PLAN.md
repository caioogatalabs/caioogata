    Interface:
    ```typescript
    export function useExperienceNavigation({
      itemCount: number;
      onExpand: (index: number) => void;
      onCollapse: (index: number) => void;
      containerRef: React.RefObject<HTMLElement | null>;
    }): {
      activeIndex: number;        // keyboard-focused row
      hoveredIndex: number;       // mouse-hovered row (-1 = none)
      highlightedIndex: number;   // whichever is active (hover takes precedence)
      setHoveredIndex: (i: number) => void;
      handleKeyDown: (e: React.KeyboardEvent) => void;
      isDimmed: (index: number) => boolean; // true if another row is highlighted
    }
    ```

    Implementation details:
    - ArrowDown/ArrowUp: move `activeIndex` through rows (wrap around)
    - Enter: toggle expand/collapse on `activeIndex` (calls `onExpand`/`onCollapse`)
    - Escape: collapse all (call `onCollapse(-1)` or similar signal)
    - `highlightedIndex`: if `hoveredIndex >= 0` use that, else `activeIndex`
    - `isDimmed(i)`: returns `highlightedIndex >= 0 && highlightedIndex !== i` (per D-21)
    - Separate `hoveredIndex` from `activeIndex` (per Phase 02 decision: independent mouse/keyboard tracking)
    - Focus management: when `activeIndex` changes via keyboard, scroll the row into view
  </action>
