    ```tsx
    'use client'

    import Image from 'next/image'
    import { useInView } from '@/hooks/useInView'
    import { Grid, GridItem } from '@/components/layout/Grid'
    import { SectionDivider } from './SectionDivider'
    import content from '@/content/en.json'
    import type { Content } from '@/content/types'

    const typedContent = content as unknown as Content
    const clientsData = typedContent.clients

    type LogoEntry = { src: string; imgClass?: string }

    /**
     * Client logo SVG map. Copied verbatim from V1 src/components/sections/Clients.tsx.
     * Do NOT import from V1 — V1 must remain self-contained for the v1.caioogata.com deploy.
     */
    const CLIENT_LOGOS: Record<string, LogoEntry> = {
      'Petrobras':   { src: '/clients/petrobras.svg' },
      'O Boticário': { src: '/clients/o-boticario.svg', imgClass: 'max-w-[130px]' },
      'Tramontina':  { src: '/clients/tramontina.svg' },
      'Sicredi':     { src: '/clients/sicredi.svg' },
      'Aché Group':  { src: '/clients/ache-group.svg' },
      'Itaú':        { src: '/clients/itau.svg' },
      'Caixa':       { src: '/clients/caixa.svg' },
      'Lacta':       { src: '/clients/lacta.svg' },
      'LG':          { src: '/clients/lg.svg', imgClass: 'max-h-[32px] max-w-[72px]' },
      'Novartis':    { src: '/clients/novartis.svg' },
      'Fila':        { src: '/clients/fila.svg' },
      'Netshoes':    { src: '/clients/netshoes.svg' },
      'Magalu':      { src: '/clients/magalu.svg' },
      'Exame':       { src: '/clients/exame.svg' },
      'Dafiti':      { src: '/clients/dafiti.svg' },
      'RocketChat':  { src: '/clients/rocketchat.svg' },
    }

    /** Desktop: 4 cols. Mobile: 2 cols. */
    const DESKTOP_COLS = 4
    const MOBILE_COLS = 2

    export function ClientsBlock() {
      const blockRef = useInView({ threshold: 0.1, once: true })
      const items = clientsData.list
      const shortDescription = clientsData.shortDescription ?? clientsData.description

      return (
        <div>
          <SectionDivider code="1.3" label="Notable Clients" />

          <div
            ref={blockRef as React.RefObject<HTMLDivElement>}
            className="px-5 md:px-8 lg:px-16 py-16 md:py-24 lg:py-32"
          >
            {/* Description in right 6-col */}
            <Grid className="!px-0 mb-12 md:mb-16">
              <GridItem span={6} tabletSpan={2} mobileSpan={4} />
              <GridItem span={6} tabletSpan={6} mobileSpan={4}>
                <p
                  className="-entrance -slide-up -a-0 text-[20px] md:text-[24px] leading-[1.4] text-text-secondary"
                  style={{ fontFamily: 'var(--font-sans)' }}
                >
                  {shortDescription}
                </p>
              </GridItem>
            </Grid>

            {/* Logo grid: 2 cols mobile, 4 cols desktop */}
            <div className="grid grid-cols-2 lg:grid-cols-4">
              {items.map((client, index) => {
                const logo = CLIENT_LOGOS[client]
                const colIndex = index % DESKTOP_COLS         // 0..3 desktop
                const rowIndex = Math.floor(index / DESKTOP_COLS) // 0..3 desktop
                const colIndexMobile = index % MOBILE_COLS    // 0..1 mobile

                // Stagger cap at 19 (CSS classes -a-0 .. -a-20).
                const stagger = Math.min(index, 19)

                return (
                  <div
                    key={client}
                    className={[
                      '-entrance -fade',
                      `-a-${stagger}`,
                      'min-h-[120px] flex items-center justify-center',
                      // Right divider: skip last column (mobile last col = 1, desktop last col = 3)
                      colIndexMobile < MOBILE_COLS - 1 ? 'border-r border-border-secondary' : '',
                      // On desktop, override mobile decision: right border on col 0,1,2 (skip col 3)
                      colIndex < DESKTOP_COLS - 1 ? 'lg:border-r lg:border-border-secondary' : 'lg:border-r-0',
                      // Bottom divider: skip first row (top divider style — matches Skills/Education pattern)
                      // We add `border-t` on rows >= 1 for both mobile and desktop.
                      rowIndex > 0 ? 'lg:border-t lg:border-border-secondary' : '',
                      // Mobile: top border on every row except the first mobile row.
                      // Mobile rows = ceil(16/2) = 8 rows; row index = floor(index/2).
                      Math.floor(index / MOBILE_COLS) > 0 ? 'border-t border-border-secondary' : '',
                    ].filter(Boolean).join(' ')}
                  >
                    {logo ? (
                      <Image
                        src={logo.src}
                        alt={client}
                        width={120}
                        height={48}
                        className={`object-contain max-h-[44px] max-w-[108px]${logo.imgClass ? ` ${logo.imgClass}` : ''}`}
                        unoptimized
                      />
                    ) : (
                      <span
                        className="font-mono text-xs text-text-tertiary text-center"
                        style={{ fontFamily: 'var(--font-mono)' }}
                      >
                        {client}
                      </span>
                    )}
                  </div>
                )
              })}
            </div>
          </div>
        </div>
      )
    }
    ```

    Concrete rules (per D-14 .. D-18):
    - `shortDescription` is read from `clientsData.shortDescription ?? clientsData.description` — defensive fallback in case Wave 1's edit got reverted; the optional `?` chain works because Wave 1 should have added `shortDescription` to types.ts (otherwise the `?? description` clause keeps build passing). Confirm via 04.2-01-SUMMARY.md.
    - Description sits in the **right** 6-col (`GridItem span={6}` after a 6-col spacer left), per D-14 ("right 6-col above the logo grid"). On mobile/tablet it occupies the full width.
    - Logo grid is a plain CSS grid `grid-cols-2 lg:grid-cols-4` — NOT a 12-col Grid. With 16 logos + 4 cols, exactly 4 rows × 4 cols on desktop. With 2 cols on mobile, exactly 8 rows × 2 cols.
    - Cell styling per D-16/D-17:
      - `min-h-[120px]` — D-17 sizing.
      - `flex items-center justify-center` — center logo.
      - NO `border` (full), NO `bg-*`, NO `shadow-*`, NO `rounded-*`.
      - Right divider on cells where `colIndex < cols - 1` — at desktop applies to cols 0/1/2 (skip last col 3); at mobile applies to col 0 only. The two responsive states are handled with `lg:border-r-0` overriding mobile's `border-r` for desktop's last column.
      - Top divider on cells where `rowIndex > 0` — at desktop applies to rows 1/2/3; at mobile applies to rows 1..7.
    - Logo image: `next/image` with `unoptimized` (static export — required), `width=120 height=48`, `object-contain max-h-[44px] max-w-[108px]` plus optional per-logo `imgClass` override (e.g., LG and O Boticário).
    - Fallback for missing logo: render the brand name in mono — should not occur in practice (all 16 are mapped) but matches V1 defensive pattern.
    - SectionDivider: `code="1.3"` `label="Notable Clients"`.
    - Stagger via `-entrance -fade -a-${stagger}` per cell, capped at 19.
    - Do NOT import from V1 (`@/components/sections/Clients`).
    - Do NOT use `next/dynamic`, no client-only lazy load (logos are tiny SVGs).

    Note on dynamic Tailwind classes: the `-a-${stagger}` classes are custom CSS (not Tailwind utilities) defined in the existing animation system, so dynamic interpolation is safe. The actual Tailwind classes (`border-r`, `border-border-secondary`, `lg:border-r-0`, `lg:border-t`) are all literal in the source.
