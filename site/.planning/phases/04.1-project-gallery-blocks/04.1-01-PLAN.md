**Main component structure:**

```typescript
export function ProjectGalleryStaggered({ section }: ProjectGalleryStaggeredProps) {
  const rows = section.rows || []
  if (rows.length === 0) return null

  return (
    <section className="py-24 px-5 md:px-8 lg:px-16">
      <div className="flex flex-col gap-y-[8px]">
        {rows.map((row, rowIdx) => (
          <div key={rowIdx} className="grid grid-cols-12 gap-x-[4px]">
            {row.images.map((item, imgIdx) => {
              const span = row.spans[imgIdx] ?? Math.floor(12 / row.images.length)
              const spanClasses = [
                MOBILE_SPAN[span] || 'col-span-12',
                TABLET_SPAN[span] || 'md:col-span-6',
                DESKTOP_SPAN[span] || 'lg:col-span-12',
              ].join(' ')

              return (
                <div key={imgIdx} className={spanClasses}>
                  <StaggeredMedia item={item} staggerIndex={imgIdx} />
                </div>
              )
            })}
          </div>
        ))}
      </div>
    </section>
  )
}
```

**Defensive fallback (per Research pitfall 5):** If `row.spans[imgIdx]` is undefined, fall back to `Math.floor(12 / row.images.length)`. Log a console.warn in development: `console.warn(\`[Gallery] Row ${rowIdx}: spans.length (${row.spans.length}) !== images.length (${row.images.length})\`)`.

**Reduced motion (per D-18):** Handled by `useScrollReveal` hook itself (it already checks `prefers-reduced-motion` and sets clipPath to fully revealed). No additional reduced motion code needed in this component.

Keep `'use client'` directive at top. Keep all existing imports. Add no new external dependencies.
